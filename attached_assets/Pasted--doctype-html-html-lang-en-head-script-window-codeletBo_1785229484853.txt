          { assessment: 'Capstone Submission', score: '—', weight: '75%', status: 'Pending' }
        ],
        attendance: { present: '23', absent: '1', rate: '96%' }
      }
    };

    // Export page functionality
    const exportSelect = document.getElementById('export-course-select');
    const exportInfoCard = document.querySelector('[data-template-id="export-info-card"]');
    const exportGradesBody = document.getElementById('export-grades-body');
    const exportJsonBtn = document.getElementById('export-json-btn');

    function updateExportDisplay(courseId) {
      const data = courseData[courseId];
      if (!data) return;

      // Update course info
      document.querySelector('[data-template-id="export-course-name"]').textContent = data.name;
      document.querySelector('[data-template-id="export-course-info"]').textContent = data.info;
      document.querySelector('[data-template-id="export-stat-current"]').textContent = data.current;
      document.querySelector('[data-template-id="export-stat-instructor"]').textContent = data.instructor;
      document.querySelector('[data-template-id="export-stat-enrolled"]').textContent = data.enrolled;

      // Update grades table
      exportGradesBody.innerHTML = '';
      data.grades.forEach(grade => {
        const row = document.createElement('tr');
        row.className = 'border-t';
        row.innerHTML = `
          <td class="p-3 font-medium text-slate-700">${grade.assessment}</td>
          <td class="p-3 font-bold ${grade.score === '—' ? 'text-slate-400' : 'text-teal-600'}">${grade.score}</td>
          <td class="p-3 text-slate-500">${grade.weight}</td>
          <td class="p-3">
            <span class="inline-block px-2 py-1 rounded-md text-xs font-semibold ${
              grade.status === 'Graded' 
                ? 'bg-emerald-100 text-emerald-700' 
                : 'bg-amber-100 text-amber-700'
            }">
              ${grade.status}
            </span>
          </td>
        `;
        exportGradesBody.appendChild(row);
      });

      // Update attendance
      document.querySelector('[data-template-id="export-attendance-present"]').textContent = data.attendance.present;
      document.querySelector('[data-template-id="export-attendance-absent"]').textContent = data.attendance.absent;
      document.querySelector('[data-template-id="export-attendance-rate"]').textContent = data.attendance.rate;
    }

    exportSelect.addEventListener('change', (e) => {
      updateExportDisplay(e.target.value);
    });

    exportJsonBtn.addEventListener('click', () => {
      const courseId = exportSelect.value;
      const data = courseData[courseId];
      const studentName = document.querySelector('[data-template-id="profile-name"]')?.textContent || 'Nguyen Van A';
      const studentId = document.querySelector('[data-template-id="cs301-student-id"]')?.textContent || 'STU001';

      const exportData = {
        student: {
          name: studentName,
          id: studentId,
          exportDate: new Date().toISOString()
        },
        course: {
          id: courseId,
          name: data.name,
          info: data.info,
          currentGrade: data.current,
          instructor: data.instructor,
          enrolled: data.enrolled
        },
        grades: data.grades,
        attendance: data.attendance
      };

      const jsonString = JSON.stringify(exportData, null, 2);
      const blob = new Blob([jsonString], { type: 'application/json' });
      const url = URL.createObjectURL(blob);
      const link = document.createElement('a');
      link.href = url;
      link.download = `${courseId}-report-${new Date().toISOString().split('T')[0]}.json`;
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
      URL.revokeObjectURL(url);

      showToast('Report exported successfully!');
    });

    // Initialize export display on first load
    if (document.readyState === 'loading') {
      document.addEventListener('DOMContentLoaded', () => {
        computeStatistics();
        updateMiniProfile();
        updateExportDisplay('cs301');
      });
    } else {
      computeStatistics();
      updateMiniProfile();
      updateExportDisplay('cs301');
    }
