    function renderAiPredictions() {
      const tbody = document.getElementById("ai-prediction-table-body");
      if (!tbody) return;
      tbody.innerHTML = "";

      const course = courseData[selectedCourse];
      const students = course.students || [];
      const assignments = course.assignments || [];
      const attendance = attendanceState[selectedCourse]?.sessions || [];

      students.forEach(student => {
        // Calculate Attendance Rate with custom rules: every 2 lates = 1 absence; excused doesn't count as absent, but we only count Present or Late as attended.
        let presentCount = 0;
        let lateCount = 0;
        let excusedCount = 0;
        let actualAbsentCount = 0;
        let totalSessions = 0;
        
        attendance.forEach(s => {
          if (s.records && s.records[student.id]) {
            totalSessions++;
            const status = s.records[student.id];
            if (status === "present") {
              presentCount++;
            } else if (status === "late") {
              lateCount++;
            } else if (status === "excused") {
              excusedCount++;
            } else if (status === "absent") {
              actualAbsentCount++;
            }
          }
        });
        
        const equivalentAbsencesFromLates = Math.floor(lateCount / 2);
        // Only Present and Late count towards attended, minus equivalent absences from lates.
        const attendedSessions = presentCount + lateCount - equivalentAbsencesFromLates;
        const attendanceRate = totalSessions > 0 ? Math.round((Math.max(0, attendedSessions) / totalSessions) * 100) : 100;

        // Calculate Predicted Final Grade (Weighted Average based on current scores)
        let weightedScore = 0;
        let totalWeight = 0;
        assignments.forEach(assign => {
          const scoreVal = student.scores[assign.id] !== undefined ? student.scores[assign.id] : 0;
          const maxPts = assign.maxPoints || 100;
          weightedScore += (scoreVal / maxPts) * assign.weight;
          totalWeight += assign.weight;
        });
        const predictedGrade = totalWeight > 0 ? (weightedScore / totalWeight) * 100 : 0;

        // At-risk if final grade < 60 OR attendance <= 80%
        const isAtRisk = predictedGrade < 60 || attendanceRate <= 80;

        const tr = document.createElement("tr");
        tr.className = "border-t border-slate-200 hover:bg-slate-50 transition-colors";

        const tdName = document.createElement("td");
        tdName.className = "p-4 font-semibold text-slate-900";
        tdName.textContent = student.name;

        const tdAttendance = document.createElement("td");
        tdAttendance.className = `p-4 font-semibold ${attendanceRate <= 80 ? "text-rose-600" : "text-slate-700"}`;
        tdAttendance.textContent = `${attendanceRate}%`;

        const tdGrade = document.createElement("td");
        tdGrade.className = `p-4 font-bold ${predictedGrade < 60 ? "text-rose-600" : "text-teal-700"}`;
        tdGrade.textContent = `${predictedGrade.toFixed(1)}% (${gradeLetter(predictedGrade)})`;

        const tdRisk = document.createElement("td");
        tdRisk.className = "p-4 text-right";

        const badge = document.createElement("span");
        if (isAtRisk) {
          badge.className = "inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-bold bg-rose-100 text-rose-800 border border-rose-200";
          badge.innerHTML = `<i data-lucide="alert-triangle" class="w-3.5 h-3.5 text-rose-700"></i> At Risk`;
        } else {
          badge.className = "inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-bold bg-emerald-100 text-emerald-800 border border-emerald-200";
          badge.innerHTML = `<i data-lucide="check-circle-2" class="w-3.5 h-3.5 text-emerald-700"></i> Good Standing`;
        }
        tdRisk.appendChild(badge);

        tr.append(tdName, tdAttendance, tdGrade, tdRisk);
        tbody.appendChild(tr);
      });

      lucide.createIcons();
    }

    function renderRankingCard() {

      const selectedLabel = document.getElementById("ranking-sort-selected-label");

      const students = course.students || [];
      const attendance = attendanceState[selectedCourse]?.sessions || [];
      const assignments = course.assignments || [];

      // 1. Populate Dropdown list dynamically based on assignments + overall metrics
      dropdownContainer.innerHTML = "";
      
      const options = [
        { value: "overall", label: "Final Grade" },
        { value: "attendance", label: "Attendance" },
        ...assignments.map(a => ({ value: a.id, label: a.name }))
      ];

      // Update current selected label display
      const currentOpt = options.find(opt => opt.value === selectedRankingCriteria) || options[0];
      if (selectedLabel) selectedLabel.textContent = currentOpt.label;

      options.forEach(opt => {
        const btn = document.createElement("button");
        btn.type = "button";
        const isCurrent = opt.value === selectedRankingCriteria;
        btn.className = `w-full text-left px-3 py-1.5 hover:bg-slate-100 dark:hover:bg-slate-700 block transition-colors ${isCurrent ? "font-bold text-teal-700 dark:text-teal-400 bg-slate-50 dark:bg-slate-800" : "text-slate-700 dark:text-slate-300"}`;
        btn.textContent = opt.label;
        btn.addEventListener("click", () => {
          selectedRankingCriteria = opt.value;
          dropdownContainer.classList.add("hidden");
          renderRankingCard();
          renderCharts();
        });
        dropdownContainer.appendChild(btn);
      });

      // 2. Setup Dropdown Trigger click toggle
      dropdownBtn.onclick = (e) => {
        e.stopPropagation();
        dropdownContainer.classList.toggle("hidden");
      };

      // 3. Compute Ranked Data based on chosen dynamic criteria
      const rankedData = students.map(student => {
        let val = 0;
        let displayVal = "";

        if (selectedRankingCriteria === "attendance") {
          let presentCount = 0;
          let lateCount = 0;
          let excusedCount = 0;
          let actualAbsentCount = 0;
          let totalSessions = 0;
          
          attendance.forEach(s => {
            if (s.records && s.records[student.id]) {
              totalSessions++;
              const status = s.records[student.id];
              if (status === "present") {
                presentCount++;
              } else if (status === "late") {
                lateCount++;
              } else if (status === "excused") {
                excusedCount++;
              } else if (status === "absent") {
                actualAbsentCount++;
              }
            }
          });
          
          const equivalentAbsencesFromLates = Math.floor(lateCount / 2);
          const attendedSessions = presentCount + lateCount + excusedCount - equivalentAbsencesFromLates;
          val = totalSessions > 0 ? Math.round((Math.max(0, attendedSessions) / totalSessions) * 100) : 100;
          displayVal = `${val}%`;
        } else if (selectedRankingCriteria === "overall") {
          let weightedScore = 0;
          let totalWeight = 0;
          assignments.forEach(assign => {

    function renderStudentDetail() {
      const detailEmpty = document.getElementById("student-detail-empty");
      const detailContent = document.getElementById("student-detail-content");
      if (!detailEmpty || !detailContent) return;

      if (!selectedStudentId) {
        detailEmpty.classList.remove("hidden");
        detailContent.classList.add("hidden");
        return;
      }

      const course = courseData[selectedCourse];
      const student = course.students.find(s => s.id === selectedStudentId);
      if (!student) {
        selectedStudentId = null;
        detailEmpty.classList.remove("hidden");
        detailContent.classList.add("hidden");
        return;
      }

      detailEmpty.classList.add("hidden");
      detailContent.classList.remove("hidden");

      // Set Badge
      const statusBadge = document.getElementById("student-detail-status");
      statusBadge.textContent = student.status;
      if (student.status === "Enrolled") {
        statusBadge.className = "px-2.5 py-1 rounded-full text-[10px] font-bold bg-emerald-100 text-emerald-800";
      } else {
        statusBadge.className = "px-2.5 py-1 rounded-full text-[10px] font-bold bg-slate-100 text-slate-600";
      }

      // Name & ID
      document.getElementById("student-detail-name").textContent = student.name;
      document.getElementById("student-detail-id").textContent = student.id;
      
      const initials = student.name.split(" ").map(w => w[0]).filter(Boolean).slice(-2).join("").toUpperCase();
      document.getElementById("student-detail-avatar").textContent = initials;

      // Calculate attendance statistics based on course sessions
      const sessions = attendanceState[selectedCourse].sessions || [];
      let presentCount = 0;
      let lateCount = 0;
      let excusedCount = 0;
      let actualAbsentCount = 0;
      let totalCount = 0;

      sessions.forEach(s => {
        if (s.records && s.records[student.id]) {
          totalCount++;
          const status = s.records[student.id];
          if (status === "present") {
            presentCount++;
          } else if (status === "late") {
            lateCount++;
          } else if (status === "excused") {
            excusedCount++;
          } else if (status === "absent") {
            actualAbsentCount++;
          }
        }
      });

      // 1) excused and absent sessions do not increase the attended count.
      // 2) every 2 late arrivals are treated as equivalent to 1 absence in computed results.
      const equivalentAbsencesFromLates = Math.floor(lateCount / 2);
      const computedAbsentCount = actualAbsentCount + equivalentAbsencesFromLates;
      
      // Attendance Rate matches the computed absences relative to total counted sessions
      // Only Present and Late count towards attended, minus equivalent absences from lates.
      const attendedSessions = presentCount + lateCount - equivalentAbsencesFromLates;
      const rate = totalCount > 0 ? Math.round((Math.max(0, attendedSessions) / totalCount) * 100) : 100;

      document.getElementById("student-detail-present").textContent = (presentCount + lateCount);
      document.getElementById("student-detail-absent").textContent = computedAbsentCount;
      document.getElementById("student-detail-rate").textContent = rate + "%";

      // Session History
      const historyContainer = document.getElementById("student-session-history");
      historyContainer.innerHTML = "";

      if (sessions.length === 0) {
        historyContainer.innerHTML = `<p class="text-slate-400 text-center text-xs py-4">No sessions recorded yet.</p>`;
      } else {
        sessions.forEach(s => {
          const status = s.records ? s.records[student.id] : null;
          const div = document.createElement("div");
          div.className = "flex items-center justify-between p-2 rounded-xl bg-slate-50 border border-slate-100 text-xs";
          
          const label = document.createElement("div");
          label.className = "min-w-0";
          const title = document.createElement("p");
          title.className = "font-medium text-slate-800 truncate";
          title.textContent = s.name;
          const date = document.createElement("p");
          date.className = "text-[10px] text-slate-400 font-mono mt-0.5";
          date.textContent = s.date;
          label.append(title, date);

          const badge = document.createElement("span");
          if (status === "present") {
            badge.className = "px-2 py-0.5 rounded-lg text-[10px] font-bold bg-green-100 text-green-800";
            badge.textContent = "Present";
          } else if (status === "absent") {
            badge.className = "px-2 py-0.5 rounded-lg text-[10px] font-bold bg-red-100 text-red-800";
            badge.textContent = "Absent";
          } else if (status === "late") {

    function renderOverview() {
      const course = courseData[selectedCourse];
      
      // Calculate dynamic pass rate based on current scores
      const totalStudents = course.students.length;
      let passingStudents = 0;
      let totalGpaPoints = 0;

      course.students.forEach(student => {
        let weightedScore = 0;
        let totalWeight = 0;
        course.assignments.forEach(assign => {
          const scoreVal = student.scores[assign.id] !== undefined ? student.scores[assign.id] : 0;
          const maxPts = assign.maxPoints || 100;
          weightedScore += (scoreVal / maxPts) * assign.weight;
          totalWeight += assign.weight;
        });
        const absoluteScore = totalWeight > 0 ? (weightedScore / totalWeight) * 100 : 0;
        if (absoluteScore >= 60) {
          passingStudents++;
        }

        // Convert percentage to standard 4.0 GPA scale
        let studentGpa = 0;
        if (absoluteScore >= 90) studentGpa = 4.0;
        else if (absoluteScore >= 85) studentGpa = 3.7;
        else if (absoluteScore >= 80) studentGpa = 3.3;
        else if (absoluteScore >= 75) studentGpa = 3.0;
        else if (absoluteScore >= 70) studentGpa = 2.7;
        else if (absoluteScore >= 65) studentGpa = 2.3;
        else if (absoluteScore >= 60) studentGpa = 2.0;
        else if (absoluteScore >= 50) studentGpa = 1.0;
        else studentGpa = 0.0;

        totalGpaPoints += studentGpa;
      });

      const computedPassRate = totalStudents > 0 ? Math.round((passingStudents / totalStudents) * 100) + "%" : "100%";
      const averageGpa = totalStudents > 0 ? (totalGpaPoints / totalStudents).toFixed(2) : "4.00";

      // Dynamically update the Workspace header metrics
      const activeHeader = document.querySelector(`.course-header[data-course-header="${selectedCourse}"]`);
      if (activeHeader) {
        const studentCountEl = activeHeader.querySelector(".header-student-count");
        const gpaAverageEl = activeHeader.querySelector(".header-gpa-average");
        if (studentCountEl) studentCountEl.textContent = totalStudents;
        if (gpaAverageEl) gpaAverageEl.textContent = averageGpa;
      }

      document.getElementById("overview-pass-rate").textContent = computedPassRate;
      document.getElementById("overview-attendance-rate").textContent = course.attendanceRate;
      document.getElementById("overview-next-item").textContent = course.nextItem;
    }

    function gradeLetter(score) {
      if (score >= 90) return "A";
      if (score >= 85) return "A-";
      if (score >= 80) return "B+";
      if (score >= 75) return "B";
      if (score >= 70) return "B-";
      if (score >= 65) return "C+";
      if (score >= 60) return "C";
      if (score >= 50) return "D";
      return "F";
    }

    function calculateGrades() {
      let valid = true;
      const rows = document.querySelectorAll("#grade-table-body tr");
      const course = courseData[selectedCourse];
      
      // Check validation
      rows.forEach(row => {
        const studentId = row.dataset.studentId;
        const inputs = [...row.querySelectorAll("input")];
        inputs.forEach((input, index) => {
          const assign = course.assignments[index];
          const maxVal = assign ? (assign.maxPoints || 100) : 100;
          const value = Number(input.value);
          const isValid = Number.isFinite(value) && value >= 0 && value <= maxVal && input.value !== "";
          input.classList.toggle("border-red-400", !isValid);
          if (!isValid) valid = false;
        });
      });

      if (!valid) {
        document.getElementById("grade-message").textContent = "Enter a score from 0 to the assignment max points in every field.";
        return;
      }

      // Check total weight sums to 100%
      const totalWeight = course.assignments.reduce((sum, a) => sum + a.weight, 0);
      if (totalWeight === 0) {
        document.getElementById("grade-message").textContent = "Please add assignments to calculate grades.";
        return;
      }

      const studentResults = [];
      rows.forEach(row => {
        const studentId = row.dataset.studentId;
        const student = course.students.find(s => s.id === studentId);
        
        let weightedScore = 0;
        course.assignments.forEach(assign => {
          const scoreVal = student.scores[assign.id] !== undefined ? student.scores[assign.id] : 0;
          const maxPts = assign.maxPoints || 100;
          
          // Calculate student's earned weight contribution
          const earnedContribution = (scoreVal / maxPts) * assign.weight;
          weightedScore += earnedContribution;
        });

        // If totalWeight is not 100, scale the final result proportionally
        const absoluteScore = (weightedScore / totalWeight) * 100;

        studentResults.push({ row, score: absoluteScore });
      });

      let maxScore = 0;
      if (activeGradeMode === "curved") {
        maxScore = Math.max(...studentResults.map(r => r.score));
      }

      studentResults.forEach(({ row, score }) => {
        let finalDisplayScore = score;
        let gradeText = "";

        if (activeGradeMode === "weighted") {
          finalDisplayScore = score;
          gradeText = gradeLetter(finalDisplayScore);
        } else if (activeGradeMode === "curved") {
          finalDisplayScore = maxScore > 0 ? (score / maxScore) * 100 : score;
          gradeText = gradeLetter(finalDisplayScore);
        } else if (activeGradeMode === "passfail") {
          finalDisplayScore = score;
          gradeText = finalDisplayScore >= 60 ? "Pass" : "Fail";
        }

        row.querySelector(".weighted-score").textContent = finalDisplayScore.toFixed(1) + "%";
        
        const letterCell = row.querySelector(".letter-grade");
        letterCell.textContent = gradeText;
        if (gradeText === "Fail" || gradeText === "F") {
          letterCell.className = "p-4 font-bold text-red-600 letter-grade";
        } else {
          letterCell.className = "p-4 font-bold text-teal-700 letter-grade";
        }
      });

      const modeLabels = {
        weighted: `Weighted grades calculated: Earned / Max × Assignment Weight (scaled to ${totalWeight}% total distribution).`,
        curved: "Curved grades calculated (scaled relative to top student's score).",
        passfail: "Pass/Fail statuses calculated (60% threshold)."
      };
      document.getElementById("grade-message").textContent = modeLabels[activeGradeMode];

      // Refresh overview whenever grades are re-calculated to sync pass rate & average GPA
      renderOverview();
      renderGlobalHomeStats();
      if (activeCourseTab === "visualizations") {
        renderCharts();
        renderRankingCard();
      } else if (activeCourseTab === "ai-prediction") {
        renderAiPredictions();
      }
    }

    function processStudentCsvFile(file) {
      const reader = new FileReader();
      reader.onload = function(e) {
        const text = e.target.result;
        const parsed = parseSimpleCSV(text);

            skippedDetails.push(`Row ${rowIndex + 2}: Student ID '${studentId}' already enrolled.`);

    function parseSimpleCSV(text) {
      const lines = text.split(/\r?\n/).map(line => line.trim()).filter(line => line.length > 0);
      if (lines.length === 0) return [];
      
      const headers = lines[0].split(",").map(h => h.trim().replace(/^["']|["']$/g, ""));
      const rows = [];
      
      for (let i = 1; i < lines.length; i++) {
        const line = lines[i];
        let values = [];
        let insideQuote = false;
        let currentVal = "";
        
        for (let charIndex = 0; charIndex < line.length; charIndex++) {
          const char = line[charIndex];
          if (char === '"' || char === "'") {
            insideQuote = !insideQuote;
          } else if (char === ',' && !insideQuote) {
            values.push(currentVal.trim().replace(/^["']|["']$/g, ""));
            currentVal = "";
          } else {
            currentVal += char;
          }
        }
        values.push(currentVal.trim().replace(/^["']|["']$/g, ""));
        
        if (values.length === headers.length) {
          const rowObj = {};
          headers.forEach((header, index) => {
            rowObj[header] = values[index];
          });
          rows.push(rowObj);
        }
      }
      return { headers, rows };
    }

    function processCsvFile(file) {
      const reader = new FileReader();
      reader.onload = function(e) {
        const text = e.target.result;
        const parsed = parseSimpleCSV(text);
        if (!parsed || !parsed.rows || parsed.rows.length === 0) {
          showImportFeedback(false, "Invalid or empty CSV format. Make sure it contains correct headers.");
          return;
        }

        const course = courseData[selectedCourse];
        let importedCount = 0;
        let skippedCount = 0;
        const skippedDetails = [];

        // Find standard headers representing names
        const nameHeader = parsed.headers.find(h => h.toLowerCase() === "student" || h.toLowerCase() === "student name" || h.toLowerCase() === "name");
        if (!nameHeader) {
          showImportFeedback(false, "Could not find a student name column (e.g. 'Student Name', 'Name', or 'Student') in the CSV.");
          return;
        }

        parsed.rows.forEach((row, rowIndex) => {
          const studentNameInRow = row[nameHeader];
          if (!studentNameInRow) {
            skippedCount++;
            skippedDetails.push(`Row ${rowIndex + 2}: Missing Student Name.`);
            return;
          }

          // Search matching student in selectedCourse
          const student = course.students.find(s => s.name.toLowerCase().trim() === studentNameInRow.toLowerCase().trim());
          if (!student) {
            skippedCount++;
            skippedDetails.push(`Row ${rowIndex + 2}: Student '${studentNameInRow}' not found in course roster.`);
            return;
          }

          // Cycle through all non-name keys in the parsed headers to find matching assignments
          let rowMatchedSomeScore = false;
          parsed.headers.forEach(header => {
            if (header === nameHeader) return;

            // Search matching assignment
            const assignment = course.assignments.find(a => a.name.toLowerCase().trim() === header.toLowerCase().trim());
            if (assignment) {
              const scoreValue = Number(row[header]);
              if (!isNaN(scoreValue)) {
                // Constrain within assignment boundaries
                const constrainedScore = Math.max(0, Math.min(assignment.maxPoints || 100, scoreValue));
                student.scores[assignment.id] = constrainedScore;
                importedCount++;
                rowMatchedSomeScore = true;
              } else {
                skippedDetails.push(`Row ${rowIndex + 2} (${header}): Invalid numeric score '${row[header]}'.`);
              }
            }
          });

          if (!rowMatchedSomeScore) {
            skippedCount++;
            skippedDetails.push(`Row ${rowIndex + 2}: No matching assignment headers matched for student '${studentNameInRow}'.`);
          }
        });

        // Re-render components with newly updated inputs
        renderGrades();
        
        // Show detailed results message
        let feedbackHtml = `
          <p class="font-bold text-slate-800 mb-2">Import Finished!</p>
          <div class="flex gap-4 mb-2">
            <span class="text-green-700 font-semibold">✓ ${importedCount} Grades imported</span>
            <span class="text-amber-700 font-semibold">⚠ ${skippedCount} Rows skipped/warned</span>
          </div>
        `;
        if (skippedDetails.length > 0) {
          feedbackHtml += `
            <div class="max-h-24 overflow-y-auto bg-slate-50 border border-slate-100 p-2 rounded text-[10px] space-y-1 text-slate-500">
              <p class="font-semibold text-slate-600">Import Logs:</p>
              ${skippedDetails.map(log => `<p>${log}</p>`).join("")}
            </div>
          `;
        }

        showImportFeedback(true, feedbackHtml);
      };
      reader.readAsText(file);
    }

    function updateOverviewAttendanceStats() {
      const course = attendanceState[selectedCourse];
      const students = courseData[selectedCourse].students || [];
      
      let totalStudentRateSum = 0;
      students.forEach(student => {
        let presentCount = 0;
        let lateCount = 0;
        let excusedCount = 0;
        let totalSessions = 0;
        
        course.sessions.forEach(s => {
          if (s.records && s.records[student.id]) {
            totalSessions++;
            const status = s.records[student.id];
            if (status === "present") presentCount++;
            else if (status === "late") lateCount++;
            else if (status === "excused") excusedCount++;
          }
        });
        
        const equivalentAbsencesFromLates = Math.floor(lateCount / 2);
        // Only Present and Late count towards attended
        const attendedSessions = presentCount + lateCount - equivalentAbsencesFromLates;
        const rate = totalSessions > 0 ? Math.round((Math.max(0, attendedSessions) / totalSessions) * 100) : 100;
        totalStudentRateSum += rate;
      });
      
      const newRate = students.length > 0 ? Math.round(totalStudentRateSum / students.length) + "%" : "100%";
      courseData[selectedCourse].attendanceRate = newRate;
      document.getElementById("overview-attendance-rate").textContent = newRate;
      
      // Update global stats whenever attendance changes
      renderGlobalHomeStats();
    }

    // Function to calculate and render global stats dynamically on the Home card view
    function renderGlobalHomeStats() {
      const courses = Object.keys(courseData);
      const totalCourses = courses.length;

      // Unique student count across all courses
      const allStudentIds = new Set();
      courses.forEach(cId => {
        courseData[cId].students.forEach(stu => {
          allStudentIds.add(stu.id);
        });
      });
      const totalUniqueStudents = allStudentIds.size;

      // Average Attendance across all courses utilizing student rates with custom logic
      let totalCourseRateSum = 0;
      courses.forEach(cId => {
        const rateStr = courseData[cId].attendanceRate || "100%";
        totalCourseRateSum += parseInt(rateStr) || 100;
      });
      const globalAttendanceRate = totalCourses > 0 ? Math.round(totalCourseRateSum / totalCourses) : 89;

      // Dynamic calculation for grades pending.
      // E.g., we define a pending grade as any student score that is exactly 0.
      let pendingGradesCount = 0;
      courses.forEach(cId => {
        courseData[cId].students.forEach(student => {
          courseData[cId].assignments.forEach(assign => {
            if (student.scores[assign.id] === undefined || student.scores[assign.id] === 0) {
              pendingGradesCount++;
            }
          });
        });
      });

      const coursesEl = document.getElementById("home-stat-courses-val");
      const studentsEl = document.getElementById("home-stat-students-val");
      const attendanceEl = document.getElementById("home-stat-attendance-val");
      const pendingEl = document.getElementById("home-stat-pending-val");

      if (coursesEl) coursesEl.textContent = totalCourses;
      if (studentsEl) studentsEl.textContent = totalUniqueStudents;
      if (attendanceEl) attendanceEl.textContent = `${globalAttendanceRate}%`;
      if (pendingEl) pendingEl.textContent = pendingGradesCount;

      // Update Profile UI stats
      const profCourses = document.getElementById("profile-courses-count");
      const profStudents = document.getElementById("profile-students-count");
      if (profCourses) profCourses.textContent = totalCourses;
      if (profStudents) profStudents.textContent = totalUniqueStudents;
    }

    function renderCharts() {
      const course = courseData[selectedCourse];
      const canvas = document.getElementById("visualization-canvas");
      if (!canvas) return;
      const ctx = canvas.getContext("2d");
      if (!ctx) return;

      // Clear Canvas
      ctx.clearRect(0, 0, canvas.width, canvas.height);

      const warning = document.getElementById("chart-warning-container");
      const canvasContainer = document.getElementById("chart-canvas-container");
      const caption = document.getElementById("chart-caption");

      if (course.students.length === 0) {
        if (warning) warning.classList.remove("hidden");
        if (canvasContainer) canvasContainer.classList.add("hidden");
        return;
      } else {
        if (warning) warning.classList.add("hidden");
        if (canvasContainer) canvasContainer.classList.remove("hidden");
      }

      // Draw helper function
      const drawGrid = (maxVal, steps) => {
        ctx.strokeStyle = "#e2e8f0";
        ctx.lineWidth = 1;
        ctx.fillStyle = "#64748b";
        ctx.font = "10px sans-serif";
        ctx.textAlign = "right";

        for (let i = 0; i <= steps; i++) {
          const val = (maxVal / steps) * i;
          const y = canvas.height - 40 - ((canvas.height - 70) / steps) * i;
          ctx.beginPath();
          ctx.moveTo(50, y);
          ctx.lineTo(canvas.width - 20, y);
          ctx.stroke();
          ctx.fillText(Math.round(val), 42, y + 3);
        }
      };

      if (activeChartType === "histogram") {
        caption.textContent = "Distribution of calculated student performance scores (0% - 100%).";
        // Calculate scores
        const scores = course.students.map(student => {
          let weightedScore = 0;
          let totalWeight = 0;
          course.assignments.forEach(assign => {
            const scoreVal = student.scores[assign.id] !== undefined ? student.scores[assign.id] : 0;
            const maxPts = assign.maxPoints || 100;
            weightedScore += (scoreVal / maxPts) * assign.weight;
            totalWeight += assign.weight;
          });
          return totalWeight > 0 ? (weightedScore / totalWeight) * 100 : 0;
        });

        // Split into 5 bins: 0-20, 21-40, 41-60, 61-80, 81-100
        const bins = [0, 0, 0, 0, 0];
        scores.forEach(s => {
          if (s <= 20) bins[0]++;
          else if (s <= 40) bins[1]++;
          else if (s <= 60) bins[2]++;
          else if (s <= 80) bins[3]++;
          else bins[4]++;
        });

        const maxBin = Math.max(...bins, 1);
        drawGrid(maxBin, maxBin);

        const binLabels = ["0-20%", "21-40%", "41-60%", "61-80%", "81-100%"];
        const barWidth = (canvas.width - 80) / 5;

        bins.forEach((count, idx) => {
          const barHeight = ((canvas.height - 70) / maxBin) * count;

    // Function to calculate and render global stats dynamically on the Home card view
    function renderGlobalHomeStats() {
      const courses = Object.keys(courseData);
      const totalCourses = courses.length;

      // Unique student count across all courses
      const allStudentIds = new Set();
      courses.forEach(cId => {
        courseData[cId].students.forEach(stu => {
          allStudentIds.add(stu.id);
        });
      });
      const totalUniqueStudents = allStudentIds.size;

      // Average Attendance across all courses
      let totalRostered = 0;
      let totalAttended = 0;
      courses.forEach(cId => {
        const sessions = attendanceState[cId]?.sessions || [];
        sessions.forEach(s => {
          if (s.records) {
            Object.values(s.records).forEach(status => {
              totalRostered++;
              if (status === "present" || status === "late" || status === "excused") {
                totalAttended++;
              }
            });
          }
        });
      });
      const globalAttendanceRate = totalRostered > 0 ? Math.round((totalAttended / totalRostered) * 100) : 89;

      // Dynamic calculation for grades pending.
      let pendingGradesCount = 0;
      courses.forEach(cId => {
        courseData[cId].students.forEach(student => {
          courseData[cId].assignments.forEach(assign => {
            if (student.scores[assign.id] === undefined || student.scores[assign.id] === 0) {
              pendingGradesCount++;
            }
          });
        });
      });

      const coursesEl = document.getElementById("home-stat-courses-val");
      const studentsEl = document.getElementById("home-stat-students-val");
      const attendanceEl = document.getElementById("home-stat-attendance-val");
      const pendingEl = document.getElementById("home-stat-pending-val");

      if (coursesEl) coursesEl.textContent = totalCourses;
      if (studentsEl) studentsEl.textContent = totalUniqueStudents;
      if (attendanceEl) attendanceEl.textContent = `${globalAttendanceRate}%`;
      if (pendingEl) pendingEl.textContent = pendingGradesCount;

      // Update Profile UI stats
      const profCourses = document.getElementById("profile-courses-count");
      const profStudents = document.getElementById("profile-students-count");
      if (profCourses) profCourses.textContent = totalCourses;
      if (profStudents) profStudents.textContent = totalUniqueStudents;
    }

        const file = profileAvatarInput.files[0];
        if (file) {
          const reader = new FileReader();
          reader.onload = function(event) {
            profileData.avatarImg = event.target.result;
            updateProfileUI();
          };
          reader.readAsDataURL(file);
        } else {
          updateProfileUI();
        }

        closeProfileEditModal();
      });
    }

    // Initialize with local values
    updateProfileUI();

    if (profileDropdownTrigger && accountMenu) {
      profileDropdownTrigger.addEventListener("click", (e) => {
        e.stopPropagation();
        const isHidden = accountMenu.classList.toggle("hidden");
        profileDropdownTrigger.setAttribute("aria-expanded", String(!isHidden));
      });

      // Click outside to dismiss dropdown menu
      document.addEventListener("click", (e) => {
        if (!profileDropdownTrigger.contains(e.target) && !accountMenu.contains(e.target)) {
          accountMenu.classList.add("hidden");
          profileDropdownTrigger.setAttribute("aria-expanded", "false");
        }
      });
    }

    if (btnSignout) {
      btnSignout.addEventListener("click", () => {
        // Show signature toast banner for a clean immediate feedback look
        const gradeMessage = document.getElementById("grade-message");
        if (gradeMessage) {
          gradeMessage.textContent = "✓ Mock Action: Signing out of current school workspace...";
          gradeMessage.className = "text-sm text-amber-600 font-semibold";
        }
        showView("home");
        accountMenu.classList.add("hidden");
        profileDropdownTrigger.setAttribute("aria-expanded", "false");
      });
    }

    lucide.createIcons();
    renderCourseCards();
    renderOverview();
    renderEnrollment();
    renderGrades();
    renderAttendance();
    renderGlobalHomeStats();
  </script>
